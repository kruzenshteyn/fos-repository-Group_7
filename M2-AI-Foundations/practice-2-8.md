**Цель работы**

Получить практические навыки разработки адаптивного агента обучения с подкреплением для управления температурой в симулированном помещении.

- определить пространство состояний, пространство действий и функцию вознаграждения;
- реализовать правиловой контроллер температуры;
- обучить RL-агента с использованием Stable Baselines3;
- сравнить RL-агента с правиловым контроллером;
- оценить комфорт, энергопотребление и суммарное вознаграждение;
- визуализировать результаты моделирования средствами Python.

**Постановка задачи**

Рассматривается одна комната с системой электрического отопления.

Температура в комнате изменяется под воздействием:

- наружной температуры;
- тепловых потерь;
- мощности нагревателя;
- случайных возмущений.

Требуемая температура составляет:

    22 °C

Агент должен поддерживать температуру около заданного значения и одновременно уменьшать расход энергии.

В каждый момент времени агент получает состояние помещения и выбирает одно из трёх действий:

    0 — нагреватель выключен;
    1 — нагреватель включён на 50 %;
    2 — нагреватель включён на 100 %.

Один шаг моделирования соответствует 10 минутам.

Продолжительность одного эпизода:

    144 шага = 24 часа.

**Шаг 7. Создание файла среды**

Откройте файл `hvac``_``env``.``py` и добавьте следующий код:

    from __future__ import annotations

    from typing import Any

    import gymnasium as gym
    import numpy as np
    from gymnasium import spaces


    class HVACEnv(gym.Env):
        """Упрощённая среда управления отоплением комнаты."""

        metadata = {"render_modes": []}

        def __init__(
            self,
            episode_length: int = 144,
            setpoint: float = 22.0,
            comfort_weight: float = 1.0,
            energy_weight: float = 0.15,
        ) -> None:
            super().__init__()

            self.episode_length = episode_length
            self.setpoint = setpoint
            self.comfort_weight = comfort_weight
            self.energy_weight = energy_weight

            # Три дискретных уровня мощности:
            # 0 %, 50 % и 100 %.
            self.action_space = spaces.Discrete(3)

            # [температура комнаты, наружная температура,
            #  заданная температура].
            self.observation_space = spaces.Box(
                low=np.array(
                    [5.0, -20.0, 18.0],
                    dtype=np.float32,
                ),
                high=np.array(
                    [40.0, 40.0, 26.0],
                    dtype=np.float32,
                ),
                dtype=np.float32,
            )

            self.room_temperature = 20.0
            self.outside_temperature = 5.0
            self.step_number = 0

            self.total_energy = 0.0
            self.total_discomfort = 0.0

        def _get_observation(self) -> np.ndarray:
            return np.array(
                [
                    self.room_temperature,
                    self.outside_temperature,
                    self.setpoint,
                ],
                dtype=np.float32,
            )

        def _calculate_outside_temperature(self) -> float:
            """Моделирует суточное изменение наружной температуры."""

            phase = 2.0 * np.pi * self.step_number / self.episode_length

            return float(5.0 + 5.0 * np.sin(phase - np.pi / 2.0))

        def reset(
            self,
            *,
            seed: int | None = None,
            options: dict[str, Any] | None = None,
        ) -> tuple[np.ndarray, dict[str, Any]]:
            super().reset(seed=seed)

            self.step_number = 0
            self.total_energy = 0.0
            self.total_discomfort = 0.0

            # Начальная температура изменяется от эпизода к эпизоду.
            self.room_temperature = float(
                self.np_random.uniform(18.0, 24.0)
            )

            self.outside_temperature = (
                self._calculate_outside_temperature()
            )

            observation = self._get_observation()

            info = {
                "room_temperature": self.room_temperature,
                "outside_temperature": self.outside_temperature,
                "energy": 0.0,
                "discomfort": abs(
                    self.room_temperature - self.setpoint
                ),
            }

            return observation, info

        def step(
            self,
            action: int,
        ) -> tuple[
            np.ndarray,
            float,
            bool,
            bool,
            dict[str, Any],
        ]:
            if not self.action_space.contains(action):
                raise ValueError(f"Недопустимое действие: {action}")

            power = float(action) / 2.0

            self.outside_temperature = (
                self._calculate_outside_temperature()
            )

            # Теплообмен с наружным воздухом.
            heat_loss = 0.08 * (
                self.outside_temperature
                - self.room_temperature
            )

            # Нагрев помещения.
            heater_effect = 1.2 * power

            # Случайные возмущения:
            # открывание двери, ветер, внутренние тепловыделения.
            noise = float(
                self.np_random.normal(0.0, 0.05)
            )

            self.room_temperature += (
                heat_loss
                + heater_effect
                + noise
            )

            discomfort = abs(
                self.room_temperature - self.setpoint
            )

            energy = power

            reward = -(
                self.comfort_weight * discomfort
                + self.energy_weight * energy
            )

            # Дополнительный штраф за сильное отклонение.
            if discomfort > 3.0:
                reward -= 2.0

            self.total_discomfort += discomfort
            self.total_energy += energy

            self.step_number += 1

            terminated = False
            truncated = self.step_number >= self.episode_length

            observation = self._get_observation()

            info = {
                "room_temperature": self.room_temperature,
                "outside_temperature": self.outside_temperature,
                "setpoint": self.setpoint,
                "power": power,
                "energy": energy,
                "discomfort": discomfort,
                "total_energy": self.total_energy,
                "total_discomfort": self.total_discomfort,
            }

            return (
                observation,
                float(reward),
                terminated,
                truncated,
                info,
            )

**Шаг 8. Анализ модели температуры**

Изменение температуры вычисляется по упрощённому выражению:

    Tnew = Troom
           + heat_loss
           + heater_effect
           + noise.

Член `heat``_``loss` приближает теплообмен помещения с наружным воздухом:

    heat_loss = 0.08 * (
        outside_temperature - room_temperature
    )

Если наружная температура ниже комнатной, значение `heat``_``loss` отрицательно и помещение охлаждается.

Нагрев определяется выражением:

    heater_effect = 1.2 * power

Чем выше выбранная мощность, тем быстрее повышается температура.

**Шаг 9. Анализ функции вознаграждения**

Основная часть функции вознаграждения имеет вид:

    Reward =
    − ошибка температуры
    − стоимость энергии.

В программе она реализована следующим образом:

>     reward = -(
>         comfort_weight * discomfort
>         + energy_weight * energy
>     )

где:

>     discomfort = |Troom − Tset|;
>     energy — относительное энергопотребление;
>     comfort_weight — значимость комфорта;
>     energy_weight — значимость экономии энергии.

Максимальное вознаграждение получается при минимальном отклонении температуры и малом энергопотреблении.

**Шаг 15. Анализ параметров обучения**

В программе используются следующие параметры:

    MlpPolicy — полносвязная нейронная сеть;

    learning_rate — скорость обучения;

    n_steps — число шагов до обновления параметров;

    batch_size — размер мини-пакета;

    gamma — коэффициент дисконтирования;

    total_timesteps — суммарное число взаимодействий со средой.

Коэффициент дисконтирования:

    gamma = 0.99

означает, что агент учитывает не только немедленное вознаграждение, но и ожидаемое будущее вознаграждение.

Дисконтированный доход можно представить как:

    G = r0 + γr1 + γ²r2 + γ³r3 + ...

При `gamma`, близком к единице, агент сильнее учитывает долгосрочные последствия своих действий.

PPO использует нейронную сеть для приближённого представления стратегии и функции ценности. В процессе обучения оценка будущего результата уточняется по данным взаимодействия со средой. Обновления основаны на различии между ожидаемыми и фактически полученными вознаграждениями, то есть используют идею временных различий.

>         seed: int,
>     ) -> EpisodeResult:
>         env = HVACEnv()
>         controller = RuleBasedController()
>

>         controller.reset()
>

>             name="Правиловой контроллер",
>             env=env,

>             initial_observation=observation,
>         )

>     def run_rl_controller(
>         model: PPO,
>         seed: int,
>     ) -> EpisodeResult:
>         env = HVACEnv()
>         observation, _ = env.reset(seed=seed)
>
>         def choose_action(obs: np.ndarray) -> int:
>             action, _ = model.predict(
>                 obs,
>                 deterministic=True,
>             )
>
>             return int(action)
>
>         return run_episode(
>             name="RL-агент PPO",
>             env=env,
>             action_function=choose_action,
>             initial_observation=observation,
>         )
>
>
>     def run_episode(
>         name: str,
>         env: HVACEnv,
>         action_function,
>         initial_observation: np.ndarray,
>     ) -> EpisodeResult:
>         observation = initial_observation
>
>         temperatures: list[float] = []
>         outside_temperatures: list[float] = []
>         powers: list[float] = []
>         rewards: list[float] = []
>         discomforts: list[float] = []
>
>         terminated = False
>         truncated = False
>
>         while not terminated and not truncated:
>             action = action_function(observation)
>
>             observation, reward, terminated, truncated, info = (
>                 env.step(action)
>             )
>
>             temperatures.append(
>                 float(info["room_temperature"])
>             )
>
>             outside_temperatures.append(
>                 float(info["outside_temperature"])
>             )
>
>             powers.append(
>                 float(info["power"])
>             )
>
>             rewards.append(
>                 float(reward)
>             )
>
>             discomforts.append(
>                 float(info["discomfort"])
>             )
>
>         temperatures_array = np.asarray(
>             temperatures,
>             dtype=np.float64,
>         )
>
>         comfort_fraction = float(
>             np.mean(
>                 (temperatures_array >= 21.0)
>                 & (temperatures_array <= 23.0)
>             )
>         )
>
>         result = EpisodeResult(
>             name=name,
>             temperatures=temperatures,
>             outside_temperatures=outside_temperatures,
>             powers=powers,
>             rewards=rewards,
>             total_reward=float(np.sum(rewards)),
>             total_energy=float(np.sum(powers)),
>             mean_discomfort=float(np.mean(discomforts)),
>             max_discomfort=float(np.max(discomforts)),
>             comfort_fraction=comfort_fraction,
>         )
>
>         env.close()
>
>         return result
>

    E = Σ power.

**Число переключений**
